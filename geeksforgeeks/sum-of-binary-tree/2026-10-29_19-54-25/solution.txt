class Solution {
  public:
   void total(int &sum,Node *root){
       if(root==NULL){
           return;
       }
       
     sum+=root->data;
      
      total(sum,root->left);
      total(sum,root->right);
       
   }
    int sumBT(Node* root) {
        // code here
        int sum=0;
        total(sum,root);
        return sum;
    }
};
